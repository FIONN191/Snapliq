file(READ "${CMAKE_SOURCE_DIR}/brand/product.json" BRAND_JSON)
string(JSON SNAPLIQ_NAME GET "${BRAND_JSON}" desktopName)
string(JSON SNAPLIQ_VERSION GET "${BRAND_JSON}" version)
file(READ "${CMAKE_SOURCE_DIR}/config/chrome-development.json" CHROME_JSON)
string(JSON CHROME_EXTENSION_ID GET "${CHROME_JSON}" extensionId)
configure_file(product.h.in product.h @ONLY)
configure_file(resources.rc.in resources.rc @ONLY)
configure_file(app.manifest app.manifest @ONLY)

target_include_directories(Snapliq PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")

target_compile_options(Snapliq PRIVATE /utf-8 /EHsc /W4)
target_link_libraries(Snapliq PRIVATE snapliq_core windowsapp d3d11 dxgi windowscodecs shcore shlwapi shell32 ole32 comctl32 dwmapi mfplat mfreadwrite mfuuid uiautomationcore user32 gdi32 advapi32)
target_link_options(Snapliq PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/app.manifest")
