cmake_minimum_required(VERSION 3.21)
project(Snapliq LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(snapliq_core core/src/geometry.cpp)
target_include_directories(snapliq_core PUBLIC core/include)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE snapliq_core)
enable_testing()
add_test(NAME geometry COMMAND core_tests)
if(WIN32)
 enable_language(RC)
 add_subdirectory(apps/windows)
endif()
